set -euo pipefail

LOGDIR=${LOGDIR:-/opt/space-diplomat/app/data/logs}
DATES=${DATES:-"20250813 20250814 20250815 20250818 20250819 20250824"}

for d in $DATES; do
  f="$LOGDIR/rl_interactions_${d}.jsonl"
  if [ ! -s "$f" ]; then
    echo "===== $d (no file) ====="
    continue
  fi

  echo "===== $d ====="
  echo "- interactions: $(wc -l < "$f")"

  echo "- actions (count):"
  jq -r '.rl.action // empty' "$f" | sort | uniq -c | sort -nr

  echo "- avg reward by action:"
  jq -r '. as $x | "\($x.rl.action // "NA") \($x.rl.reward // 0)"' "$f" | awk '{sum[$1]+=$2; n[$1]++} END{for(k in sum) printf "  %s %.4f (n=%d)\n",k,sum[k]/n[k],n[k]}' | sort

  echo "- avg ΔPenbol (joy, anger):"
  jq -r '[(.affect.penbolAfter.joy //0) - (.affect.penbolBefore.joy //0), (.affect.penbolAfter.anger //0) - (.affect.penbolBefore.anger //0)] | @tsv' "$f" | awk -F'\t' '{jy+=$1; ag+=$2; n++} END{if(n) printf "  joy=%.4f anger=%.4f (n=%d)\n", jy/n, ag/n, n; else print "  (no rows)"}'

  echo "- avg ΔCurrent alien (joy, anger):"
  jq -r '[(.affect.currentAfter.joy //0) - (.affect.currentBefore.joy //0), (.affect.currentAfter.anger //0) - (.affect.currentBefore.anger //0)] | @tsv' "$f" | awk -F'\t' '{jy+=$1; ag+=$2; n++} END{if(n) printf "  joy=%.4f anger=%.4f (n=%d)\n", jy/n, ag/n, n; else print "  (no rows)"}'

  echo "- top states visited:"
  jq -r '.rl.state // .rl.state_key // .stateKey // empty' "$f" | sort | uniq -c | sort -nr | head -10
  echo
done
